# 2 Sets v1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the v1 of "2 Sets", a local-first PWA to log weighted calisthenics sessions with the Barseagle 2 Set Method, with progression suggestions, rest timer, history, charts and settings.

**Architecture:** Vanilla HTML/CSS/ES-modules, no build step and no runtime dependencies. Pure domain logic lives in `js/model.js` and is unit-tested with `node --test`. Persistence is IndexedDB behind a tiny promise wrapper (`js/db.js`). Screens are ES modules that render into a single `<main>` and are switched by a hash router. A service worker gives offline shell caching and notification display.

**Tech Stack:** HTML5, CSS custom properties, ES modules, IndexedDB, Service Worker, Web Audio, Screen Wake Lock, Notifications API, Node 24 built-in test runner.

**Spec:** `docs/superpowers/specs/2026-09-24-2sets-design.md`

## Global Constraints

- No build step, no npm runtime dependencies. `package.json` exists only for `npm test` → `node --test tests/`.
- All UI copy in rioplatense Spanish (vos). Block names: Ignis, Aproximación, Duo, Secundus, Extra.
- Rest timer always clamped to 180..420 seconds.
- Progression targets: dips `[10, 15]`, pullups `[8, 12]`, second exercises range `[12, 15]`.
- Loads: `{ mode: 'weight' | 'bodyweight' | 'band', kg, bandId }`; `totalLoad = bw + kg - assistKg`.
- Colors from spec section 10; fonts Nunito + Cormorant Garamond with system fallback.
- Mobile first: 16 px gutter, 48 px tap targets, no horizontal scroll, dark mode via `prefers-color-scheme` plus manual override.
- Data never leaves the device. Export/import JSON carries `app: '2sets'` and `schemaVersion: 1`.

## Review Focus

1. A set saved with `reps: 0` or a load `kg` that is `NaN`/negative must not create PRs or break suggestions → tests in Task 3 and Task 4 (`ignores sets with reps 0`, `treats NaN kg as 0`).
2. Rest timer state restored after a reload when `restEndAt` is in the past must fire once, not loop → test in Task 6 (`shouldFire fires once`).
3. Import of a JSON from a different app or a future schema must be rejected before clearing data → tests in Task 5 (`validateExport rejects`).
4. `suggestMain` for set 2 when set 1 just moved from bodyweight to weight must seed 80% and never suggest more than set 1 → test in Task 3 (`set2 seed never exceeds set1`).
5. Bodyweight missing (user never entered it) must still allow logging; totals then equal added kg → test in Task 2 (`totalLoad without bodyweight`).

```js
export function el(tag, attrs = {}, ...children) {
  const node = document.createElement(tag);
  for (const [k, v] of Object.entries(attrs || {})) {
    if (k === 'class') node.className = v;
    else if (k === 'dataset') Object.assign(node.dataset, v);
    else if (k.startsWith('on') && typeof v === 'function') node.addEventListener(k.slice(2), v);
    else if (v !== null && v !== undefined && v !== false) node.setAttribute(k, v === true ? '' : v);
  }
  for (const c of children.flat()) {
    if (c === null || c === undefined || c === false) continue;
    node.append(c instanceof Node ? c : document.createTextNode(String(c)));
  }
  return node;
}
```

```js
const usable = (s) => s && s.done !== false && Number(s.reps) > 0;
function suggestSlot(ex, s, target, inc) {
  if (!usable(s)) return { load: { mode: 'bodyweight', kg: 0, bandId: null }, hint: `Encontrá tu peso perfecto: el que te haga fallar cerca de ${target} reps.` };
  const { mode, kg = 0, bandId = null } = s.load; const reps = Number(s.reps);
  if (mode === 'weight') {
    if (reps >= target) return { load: { mode, kg: kg + inc, bandId: null }, hint: `Llegaste a ${reps}. Subí a ${kg + inc} kg.` };
    return { load: { mode, kg, bandId: null }, hint: `Mismo peso: buscá ${target} reps.` };
  }
  if (mode === 'bodyweight') {
    if (reps >= target) return { load: { mode: 'weight', kg: inc, bandId: null }, hint: `Pasá a lastre: +${inc} kg.` };
    return { load: { mode, kg: 0, bandId: null }, hint: `Peso corporal: buscá ${target} reps.` };
  }
  if (reps >= target) return { load: { mode: 'band', kg: 0, bandId }, hint: 'Probá una banda más liviana o peso corporal.' };
  return { load: { mode: 'band', kg: 0, bandId }, hint: `Misma banda: buscá ${target} reps.` };
}
export function suggestMain(exerciseId, lastBlock, opts = {}) {
  const inc = Number(opts.incrementKg) || 5;
  const [t1, t2] = EXERCISES[exerciseId].targets;
  const sets = lastBlock?.sets || [];
  const s1 = suggestSlot(exerciseId, sets[0], t1, inc);
  let s2 = suggestSlot(exerciseId, sets[1], t2, inc);
  if (s1.load.mode === 'weight' && sets[1]?.load?.mode !== 'weight') {
    const seed = Math.max(2.5, roundDown2_5(s1.load.kg * 0.8));
    s2 = { load: { mode: 'weight', kg: Math.min(seed, s1.load.kg), bandId: null }, hint: `Serie 2 al 80% de la serie 1: ${Math.min(seed, s1.load.kg)} kg.` };
  }
  const prev = opts.previousBlock?.sets?.[0], last = sets[0];
  const switchTo2_5 = inc === 5 && usable(prev) && usable(last) && prev.load.mode === 'weight' && last.load.mode === 'weight' && last.load.kg > prev.load.kg && (prev.reps - last.reps) >= 3;
  return { sets: [s1, s2], switchTo2_5 };
}
export function suggestSecond(exerciseId, lastBlock, opts = {}) {
  const inc = Number(opts.incrementKg) || 2.5; const [lo, hi] = EXERCISES[exerciseId].range;
  const sets = (lastBlock?.sets || []).filter(usable);
  if (sets.length < 2) return { load: { mode: 'bodyweight', kg: 0, bandId: null }, hint: `Arrancá sin carga, buscá ${lo} a ${hi} reps al fallo.` };
  const ref = sets[0].load; const allHi = sets.every(s => Number(s.reps) >= hi);
  if (!allHi) return { load: { ...ref }, hint: `Mantené: buscá ${hi} en las dos series.` };
  if (ref.mode === 'weight') return { load: { mode: 'weight', kg: ref.kg + inc, bandId: null }, hint: `Las dos series a ${hi}. Subí a ${ref.kg + inc} kg.` };
  if (ref.mode === 'bodyweight') return { load: { mode: 'weight', kg: inc, bandId: null }, hint: `Pasá a lastre: +${inc} kg.` };
  return { load: { ...ref }, hint: 'Probá una banda más liviana o sin banda.' };
}
export function nextDay(last) { return last?.day === 'pull' ? 'push' : 'pull'; }
```

```js
export function computePRs(sessions, opts = {}) {

    for (const { exerciseId, slot, set } of sessionSets(s)) {
      if (!set || set.done === false || !(Number(set.reps) > 0)) continue;
      const m = metricsOf(set, bw, opts.bandsById || {});
      const slotPrs = ((prs[exerciseId] ||= {})[slot] ||= { maxAdded: null, maxTotal: null, maxE1RM: null, maxBwReps: null });
      for (const [type, value] of Object.entries(m)) {
        if (value === null || value === undefined) continue;
        if (!slotPrs[type] || value > slotPrs[type].value) slotPrs[type] = { value, date: s.date };
      }
    }
  }
  return prs;
}
export function detectNewPRs(prsBefore, session, opts = {}) {

    if (!set || set.done === false || !(Number(set.reps) > 0)) continue;

      if (value === null || value === undefined || value <= 0) continue;
      if (!prev[type] || value > prev[type].value) out.push({ exerciseId, slot, type, value, prev: prev[type]?.value ?? null });
    }
  }
  return out;
}
export function milestones(sessions, opts = {}) {
  const out = {};
  for (const [ex, targetKg] of Object.entries(MILESTONES)) {
    let bestKg = 0;
    for (const s of sessions.filter(isDone)) {
      const b = s.blocks?.main; if (b?.exerciseId !== ex) continue; const set = b.sets?.[0];
      if (set?.load?.mode === 'weight' && Number(set.reps) >= 10) bestKg = Math.max(bestKg, Number(set.load.kg) || 0);
    }
    out[ex] = { targetKg, bestKg, pct: Math.min(100, Math.round(bestKg / targetKg * 100)) };
  }
  return out;
}
export function seriesFor(sessions, exerciseId, slot, metric, opts = {}) {

    const hit = sessionSets(s).find(x => x.exerciseId === exerciseId && x.slot === slot);
    if (!hit || !(Number(hit.set.reps) > 0)) continue;
    const total = totalLoad(hit.set, bw, opts.bandsById || {});
    const value = metric === 'added' ? addedKg(hit.set.load, opts.bandsById || {}) : metric === 'total' ? total
      : metric === 'e1rm' ? epley1RM(total, Number(hit.set.reps)) : Number(hit.set.reps);
    pts.push({ date: s.date, value });
  }
  return pts;
}
```

```js
const VERSION = 'v1.0.0'; const CACHE = `2sets-${VERSION}`;
const SHELL = ['./', './index.html', './manifest.webmanifest', './css/tokens.css', './css/app.css', './js/app.js', './js/ui.js', './js/db.js', './js/model.js', './js/templates.js', './js/timer.js', './js/charts.js', './js/quotes.js', './js/setrow.js', './js/screens/home.js', './js/screens/session.js', './js/screens/history.js', './js/screens/progress.js', './js/screens/settings.js', './assets/icons/icon.svg', './assets/icons/icon-192.png', './assets/icons/icon-512.png'];
self.addEventListener('install', e => e.waitUntil(caches.open(CACHE).then(c => c.addAll(SHELL)).then(() => self.skipWaiting())));
self.addEventListener('activate', e => e.waitUntil(caches.keys().then(keys => Promise.all(keys.filter(k => k !== CACHE).map(k => caches.delete(k)))).then(() => self.clients.claim())));
self.addEventListener('fetch', e => { const url = new URL(e.request.url);
  if (url.origin === location.origin) e.respondWith(caches.match(e.request).then(r => r || fetch(e.request)));
  else e.respondWith(fetch(e.request).then(r => { const copy = r.clone(); caches.open(CACHE).then(c => c.put(e.request, copy)); return r; }).catch(() => caches.match(e.request))); });
self.addEventListener('notificationclick', e => { e.notification.close(); e.waitUntil(self.clients.matchAll({ type: 'window' }).then(cs => cs[0] ? cs[0].focus() : self.clients.openWindow('./'))); });
```
